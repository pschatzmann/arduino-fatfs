131 bool begin(bool mode1bit, int max_freq_khz, sdmmc_host_t host_cfg,
132 sdmmc_slot_config_t slot_cfg) {
133 // Store configurations
134 host_config = host_cfg;
135 slot_config = slot_cfg;
136
137 // Set bus width
138 if (mode1bit) {
139 slot_config.width = 1;
140 } else {
141 slot_config.width = 4;
142 }
143
144 // Set maximum frequency
145 host_config.max_freq_khz = max_freq_khz;
146
147 // host_config.flags &= ~SDMMC_HOST_FLAG_DDR;
148
149 // Clean up any previous initialization state
150 // (in case SDMMC was used before and not properly cleaned up)
151 sdmmc_host_deinit();
152
153 // Initialize host peripheral
154 esp_err_t err = sdmmc_host_init();
155 if (err != ESP_OK) {
156 ESP_LOGE("Esp32SdmmcIO", "sdmmc_host_init failed: 0x%x", err);
157 stat = STA_NOINIT;
158 return false;
159 }
160
161 // Initialize slot (ESP32 only has SDMMC_HOST_SLOT_1)
162 err = sdmmc_host_init_slot(SDMMC_HOST_SLOT_1, &slot_config);
163 if (err != ESP_OK) {
164 ESP_LOGE("Esp32SdmmcIO", "sdmmc_host_init_slot failed: 0x%x", err);
165 sdmmc_host_deinit();
166 stat = STA_NOINIT;
167 return false;
168 }
169
170 // Allocate card structure
171 if (card == nullptr) {
172 card = (sdmmc_card_t*)malloc(sizeof(sdmmc_card_t));
173 if (card == nullptr) {
174 sdmmc_host_deinit();
175 stat = STA_NOINIT;
176 return false;
177 }
178 }
179 memset(card, 0, sizeof(sdmmc_card_t));
180
181 // Probe and initialize card
182 card->host = host_config;
183 err = sdmmc_card_init(&host_config, card);
184 if (err != ESP_OK) {
185 free(card);
186 card = nullptr;
187 sdmmc_host_deinit();
188 stat = STA_NODISK;
189 return false;
190 }
191
192 // Determine card type flags
193 CardType = 0;
194 if (card->is_mmc) {
195 CardType = CT_MMC;
196 } else {
197 // SD card
198 if (card->ocr & SD_OCR_SDHC_CAP) {
199 CardType = CT_SD2 | CT_BLOCK; // SDHC/SDXC
200 } else {
201 CardType = CT_SD2; // SDSC
202 }
203 }
204
205 stat = STA_CLEAR;
206 return true;
207 }

231 DSTATUS disk_initialize(BYTE drv) override {
232 if (drv != 0) return STA_NOINIT;
233
234 // If already initialized, return current status
235 if (stat == STA_CLEAR) return stat;
236
237 // Try to initialize with stored settings or defaults
238 if (auto_init && host_config.flags == 0) {
239 // Constructor with mode1bit and freq was used
240 if (begin(init_mode1bit, init_max_freq_khz)) {
241 return STA_CLEAR;
242 }
243 } else if (auto_init && host_config.flags != 0) {
244 // Constructor with full config was used
245 if (begin(init_mode1bit, init_max_freq_khz, host_config, slot_config)) {
246 return STA_CLEAR;
247 }
248 } else {
249 // Default constructor was used, try with defaults
250 if (begin()) {
251 return STA_CLEAR;
252 }
253 }
254
255 return stat;
256 }

317#if FF_IO_USE_IOCTL
318 DRESULT disk_ioctl(BYTE drv, ioctl_cmd_t cmd, void* buff) override {
319 if (drv != 0) return RES_PARERR;
320 if (stat & STA_NOINIT) return RES_NOTRDY;
321
322 DRESULT res = RES_ERROR;
323
324 switch (cmd) {
325 case CTRL_SYNC:
326 // SD_MMC handles synchronization internally
327 res = RES_OK;
328 break;
329
330 case GET_SECTOR_COUNT:
331 if (buff && card) {
332 // Get total sectors from card capacity
333 uint64_t capacity =
334 ((uint64_t)card->csd.capacity) * card->csd.sector_size;
335 *(DWORD*)buff = (DWORD)(capacity / 512);
336 res = RES_OK;
337 }
338 break;
339
340 case GET_SECTOR_SIZE:
341 if (buff) {
342 // SD cards use 512-byte sectors for compatibility
343 *(WORD*)buff = 512;
344 res = RES_OK;
345 }
346 break;
347
348 case GET_BLOCK_SIZE:
349 if (buff && card) {
350 // Return erase block size in sectors
351 // Use the card's erase sector size if available
352 uint32_t erase_size = card->csd.sector_size;
353 if (erase_size < 512) erase_size = 512;
354 *(DWORD*)buff = erase_size / 512;
355 if (*(DWORD*)buff == 0) *(DWORD*)buff = 1;
356 res = RES_OK;
357 }
358 break;
359
360 case CTRL_TRIM:
361 // TRIM/ERASE not implemented via SD_MMC library
362 res = RES_OK; // Return OK but don't actually trim
363 break;
364
365 case MMC_GET_TYPE:
366 if (buff) {
367 *(BYTE*)buff = CardType;
368 res = RES_OK;
369 }
370 break;
371
372 case MMC_GET_CSD:
373 if (buff && card) {
374 // Copy CSD register data
375 memcpy(buff, &card->csd, sizeof(card->csd));
376 res = RES_OK;
377 }
378 break;
379
380 case MMC_GET_CID:
381 if (buff && card) {
382 // Copy CID register data
383 memcpy(buff, &card->cid, sizeof(card->cid));
384 res = RES_OK;
385 }
386 break;
387
388 case MMC_GET_OCR:
389 if (buff && card) {
390 // Copy OCR register
391 *(DWORD*)buff = card->ocr;
392 res = RES_OK;
393 }
394 break;
395
396 case MMC_GET_SDSTAT:
397 if (buff && card) {
398 // SD Status is only defined for SD cards (ACMD13)
399 if (card->is_mmc) {
400 res = RES_PARERR;
401 } else if (read_sd_status(buff) == ESP_OK) {
402 res = RES_OK;
403 } else {
404 res = RES_ERROR;
405 }
406 }
407 break;
408
409 default:
410 res = RES_PARERR;
411 break;
412 }
413
414 return res;
415 }

462 esp_err_t read_sd_status(void* buff) {
463 sdmmc_command_t cmd;
464
465 // CMD55: next command is an application specific command
466 memset(&cmd, 0, sizeof(cmd));
467 cmd.opcode = MMC_APP_CMD;
468 cmd.arg = (uint32_t)card->rca << 16;
469 cmd.flags = SCF_CMD_AC | SCF_RSP_R1;
470 esp_err_t err = card->host.do_transaction(card->host.slot, &cmd);
471 if (err != ESP_OK) return err;
472 if (cmd.error != ESP_OK) return cmd.error;
473
474 // ACMD13: read 64 bytes of SD status as a single data block
475 memset(&cmd, 0, sizeof(cmd));
476 cmd.opcode = SD_APP_SD_STATUS;
477 cmd.arg = 0;
478 cmd.flags = SCF_CMD_ADTC | SCF_RSP_R1;
479 cmd.data = buff;
480 cmd.datalen = 64;
481 cmd.blklen = 64;
482 err = card->host.do_transaction(card->host.slot, &cmd);
483 if (err != ESP_OK) return err;
484 return cmd.error;
485 }
