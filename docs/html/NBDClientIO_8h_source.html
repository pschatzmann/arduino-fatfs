116 DRESULT disk_ioctl(BYTE drv, ioctl_cmd_t cmd, void* buff) override {
117 if (drv != 0) return RES_PARERR;
118 if (!client.isConnected()) return RES_NOTRDY;
119
120 switch (cmd) {
121 case CTRL_SYNC:
122 if (client.canFlush() && !client.flush()) return RES_ERROR;
123 return RES_OK;
124
125 case GET_SECTOR_COUNT:
126 if (!buff) return RES_PARERR;
127 *(LBA_t*)buff = (LBA_t)(client.size() / 512);
128 return RES_OK;
129 case GET_SECTOR_SIZE:
130 if (!buff) return RES_PARERR;
131 *(WORD*)buff = 512;
132 return RES_OK;
133
134 case GET_BLOCK_SIZE:
135 if (!buff) return RES_PARERR;
136 *(DWORD*)buff = client.preferredBlockSize() / 512;
137 if (*(DWORD*)buff == 0) *(DWORD*)buff = 1;
138 return RES_OK;
139
140 case CTRL_TRIM:
141 if (!buff) return RES_PARERR;
142 if (!client.canTrim()) return RES_OK; // nothing to do
143 {
144 LBA_t* range = (LBA_t*)buff; // [first sector, last sector]
145 uint64_t offset = (uint64_t)range[0] * 512;
146 uint64_t len = ((uint64_t)range[1] - range[0] + 1) * 512;

148 }

167 bool readRange(uint64_t offset, BYTE* data, size_t len) {
168 size_t max = maxChunk();

170 size_t n = len < max ? len : max;
171 if (!client.read(offset, data, n)) return false;
172 offset += n;
173 data += n;
174 len -= n;
175 }
