29 FRESULT mount(FatFs& fs, BYTE pdrv = 0) override {
30 FRESULT rc = FR_OK;
31 for (size_t j = 0; j < io_vector.size(); j++) {
32 rc = io_vector[j]->mount(fs, (BYTE)j);
33 if (rc != FR_OK) break;
34 }
35 return rc;
36 }

39 FRESULT un_mount(FatFs& fs, BYTE pdrv = 0) override {
40 FRESULT result = FR_OK;
41 for (size_t j = 0; j < io_vector.size(); j++) {
42 auto rc = io_vector[j]->un_mount(fs, (BYTE)j);
43 if (rc != FR_OK) result = rc;
44 }
45 return result;
46 }

48 DSTATUS disk_initialize(BYTE pdrv) override {
49 if (pdrv >= io_vector.size()) return STA_NODISK;
50 return io_vector[pdrv]->disk_initialize(0);

52 DSTATUS disk_status(BYTE pdrv) override {
53 if (pdrv >= io_vector.size()) return STA_NODISK;

56 DRESULT disk_read(BYTE pdrv, BYTE* buff, LBA_t sector, UINT count) override {
